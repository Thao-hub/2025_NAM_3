var hanoiFc = ee.FeatureCollection(
  "projects/vernal-tracer-472104-c0/assets/Ha_Noi_tinh_thanh_34"
);

var region = hanoiFc.geometry();

Map.centerObject(region, 9);

  hanoiFc.style({

    fillColor: '00000000',
    width: 2
  }),

  'Hanoi boundary'

// =====================
// 2) THỜI GIAN
// =====================
var startDate = '2025-03-01';
var endDate   = '2025-10-01';
var maxCloud  = 20;


// =====================
// 3) CLOUD MASK
// =====================
function maskL89(img){

  var qa = img.select('QA_PIXEL');

  var cloud  = qa.bitwiseAnd(1 << 3).neq(0);
  var shadow = qa.bitwiseAnd(1 << 4).neq(0);
  var cirrus = qa.bitwiseAnd(1 << 2).neq(0);
  var snow   = qa.bitwiseAnd(1 << 5).neq(0);

  var mask = cloud.or(shadow).or(cirrus).or(snow).not();

  return img.updateMask(mask);
}


// =====================
// 4) TÍNH LST NDVI NDBI
// =====================
function addIndices(img){

  var lstK = img.select('ST_B10')
    .multiply(0.00341802)
    .add(149.0);

  var lstC = lstK.subtract(273.15)
    .rename('LST_C');

  var ndvi = img.normalizedDifference(['SR_B5','SR_B4'])
    .rename('NDVI');

  var ndbi = img.normalizedDifference(['SR_B6','SR_B5'])
    .rename('NDBI');

  return img.addBands([lstC, ndvi, ndbi]);
}


// =====================
// 5) LANDSAT 8 + 9
// =====================
var l8 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
  .filterDate(startDate,endDate)
  .filterBounds(region)
  .filter(ee.Filter.lte('CLOUD_COVER', maxCloud))
  .map(maskL89)
  .map(addIndices);

var l9 = ee.ImageCollection("LANDSAT/LC09/C02/T1_L2")
  .filterDate(startDate,endDate)
  .filterBounds(region)
  .filter(ee.Filter.lte('CLOUD_COVER', maxCloud))
  .map(maskL89)
  .map(addIndices);

var col = l8.merge(l9);

print('L8 scenes:',l8.size());
print('L9 scenes:',l9.size());
print('Total scenes:',col.size());

var comp2025 = col
  .select(['LST_C','NDVI','NDBI'])
  .median()
  .clip(region);


// =====================
// 6) BUILT / VEG MASK
// =====================
var dw2025 = ee.ImageCollection('GOOGLE/DYNAMICWORLD/V1')
  .filterDate(startDate,endDate)
  .filterBounds(region);

var dwLabel2025 = dw2025
  .select('label')
  .mode()
  .clip(region);


// BUILT
var builtMask2025 = dwLabel2025
  .eq(6)
  .selfMask()
  .rename('BuiltMask')
  .clip(region);


// VEGETATION
var vegMask2025 = dwLabel2025
  .eq(1)
  .or(dwLabel2025.eq(2))
  .or(dwLabel2025.eq(4))
  .or(dwLabel2025.eq(5))
  .selfMask()
  .rename('VegMask')
  .clip(region);


// ADD BANDS
var out2025 = comp2025
  .addBands(builtMask2025)
  .addBands(vegMask2025)
  .clip(region);


// =====================
// 7) SUHI
// =====================
var lst2025 = out2025.select('LST_C');

var meanBuilt2025 = ee.Number(
  lst2025.updateMask(builtMask2025).reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: region,
    scale: 30,
    maxPixels: 1e13,
    bestEffort:true
  }).get('LST_C')
);

var meanVeg2025 = ee.Number(
  lst2025.updateMask(vegMask2025).reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: region,
    scale: 30,
    maxPixels: 1e13,
    bestEffort:true
  }).get('LST_C')
);

var suhiIntensity2025 = meanBuilt2025.subtract(meanVeg2025);

var suhi2025 = lst2025
  .subtract(meanVeg2025)
  .rename('SUHI_C');

print('Mean LST Built:',meanBuilt2025);
print('Mean LST Veg:',meanVeg2025);
print('SUHI intensity:',suhiIntensity2025);

out2025 = out2025.addBands(suhi2025);
