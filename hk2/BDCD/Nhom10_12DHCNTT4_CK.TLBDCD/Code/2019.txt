// =====================
// 2) TH?I GIAN
// =====================
var startDate = '2019-03-01';
var endDate   = '2019-10-01';
var maxCloud  = 20;


// =====================
// 3) CLOUD MASK
// =====================
function maskL89(img){

  var qa = img.select('QA_PIXEL');

  var cloud  = qa.bitwiseAnd(1 << 3).neq(0);
  var shadow = qa.bitwiseAnd(1 << 4).neq(0);
  var cirrus = qa.bitwiseAnd(1 << 2).neq(0);
  var snow   = qa.bitwiseAnd(1 << 5).neq(0);

  var mask = cloud.or(shadow).or(cirrus).or(snow).not();

  return img.updateMask(mask);
}


// =====================
// 4) TÍNH LST NDVI NDBI
// =====================
function addIndices(img){

  var lstK = img.select('ST_B10')
    .multiply(0.00341802)
    .add(149.0);

  var lstC = lstK.subtract(273.15)
    .rename('LST_C');

  var ndvi = img.normalizedDifference(['SR_B5','SR_B4'])
    .rename('NDVI');

  var ndbi = img.normalizedDifference(['SR_B6','SR_B5'])
    .rename('NDBI');

  return img.addBands([lstC, ndvi, ndbi]);
}


// =====================
// 5) LANDSAT 8 (2019)
// =====================
var l8 = ee.ImageCollection("LANDSAT/LC08/C02/T1_L2")
  .filterDate(startDate,endDate)
  .filterBounds(region)
  .filter(ee.Filter.lte('CLOUD_COVER', maxCloud))
  .map(maskL89)
  .map(addIndices);

print('2019 scenes:',l8.size());

var comp2019 = l8
  .select(['LST_C','NDVI','NDBI'])
  .median()
  .clip(region);


// =====================
// 6) BUILT / VEG MASK
// =====================
var dw2019 = ee.ImageCollection('GOOGLE/DYNAMICWORLD/V1')
  .filterDate(startDate,endDate)
  .filterBounds(region);

var dwLabel2019 = dw2019
  .select('label')
  .mode()
  .clip(region);


// BUILT
var builtMask2019 = dwLabel2019
  .eq(6)
  .selfMask()
  .rename('BuiltMask')
  .clip(region);


// VEGETATION
var vegMask2019 = dwLabel2019
  .eq(1)
  .or(dwLabel2019.eq(2))
  .or(dwLabel2019.eq(4))
  .or(dwLabel2019.eq(5))
  .selfMask()
  .rename('VegMask')
  .clip(region);


// ADD BANDS
var out2019 = comp2019
  .addBands(builtMask2019)
  .addBands(vegMask2019)
  .clip(region);


// =====================
// 7) SUHI
// =====================
var lst2019 = out2019.select('LST_C');

var meanBuilt2019 = ee.Number(
  lst2019.updateMask(builtMask2019).reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: region,
    scale: 30,
    maxPixels: 1e13,
    bestEffort:true
  }).get('LST_C')
);

var meanVeg2019 = ee.Number(
  lst2019.updateMask(vegMask2019).reduceRegion({
    reducer: ee.Reducer.mean(),
    geometry: region,
    scale: 30,
    maxPixels: 1e13,
    bestEffort:true
  }).get('LST_C')
);

var suhiIntensity2019 = meanBuilt2019.subtract(meanVeg2019);

var suhi2019 = lst2019
  .subtract(meanVeg2019)
  .rename('SUHI_C');

print('2019 mean LST Built:',meanBuilt2019);
print('2019 mean LST Veg:',meanVeg2019);
print('2019 SUHI intensity:',suhiIntensity2019);

out2019 = out2019.addBands(suhi2019);
